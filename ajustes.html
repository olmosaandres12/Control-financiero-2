<script>

if (!sessionStorage.getItem('olvision_auth')) window.location.href = 'index.html';

const { createClient } = supabase;

const _sb = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);

function salir() { sessionStorage.removeItem('olvision_auth'); window.location.href = 'index.html'; }


// ── Estado ────────────────────────────────────────────────

const estado = {

marcas_armazon: [],

marcas_lentes_contacto: [],

tratamientos_monofocal: [],

tratamientos_bifocal: [],

tratamientos_ocupacional: [],

tratamientos_progresivo: [],

tratamientos_tenido: []

};


const mapaUI = {

marcas_armazon: { tags: 'tags-armazon', count: 'count-armazon' },

marcas_lentes_contacto: { tags: 'tags-contacto', count: 'count-contacto' },

tratamientos_monofocal: { tags: 'tags-monofocal', count: null },

tratamientos_bifocal: { tags: 'tags-bifocal', count: null },

tratamientos_ocupacional: { tags: 'tags-ocupacional', count: null },

tratamientos_progresivo: { tags: 'tags-progresivo', count: null },

tratamientos_tenido: { tags: 'tags-tenido', count: null }

};


// ── Carga ─────────────────────────────────────────────────

async function cargarTodo() {

const { data, error } = await _sb

.from('configuracion')

.select('clave,valor')

.in('clave', Object.keys(estado));

if (error) { mostrarToast('❌ Error al cargar'); return; }

data.forEach(row => {

try { estado[row.clave] = JSON.parse(row.valor) || []; }

catch { estado[row.clave] = []; }

renderTags(row.clave);

});

}


// ── Render tags ───────────────────────────────────────────

function renderTags(clave) {

const ui = mapaUI[clave];

if (!ui) return;

const container = document.getElementById(ui.tags);

const arr = estado[clave];

const viewer = isViewer();

container.innerHTML = arr.length === 0

? '<span class="empty-tags">Sin opciones todavía</span>'

: arr.map((item, i) => `

<span class="tag-chip">

${escH(item)}

${viewer ? '' : `<button class="remove-tag" onclick="eliminarItem('${clave}',${i})">×</button>`}

</span>`).join('');

if (ui.count) document.getElementById(ui.count).textContent = arr.length;

}


// ── Agregar ───────────────────────────────────────────────

async function agregarItem(clave, inputSuffix) {

if (isViewer()) return;

const input = document.getElementById('input-' + inputSuffix);

const valor = input.value.trim();

if (!valor) return;

if (estado[clave].some(v => v.toLowerCase() === valor.toLowerCase())) {

mostrarToast('⚠️ Ya existe esa opción');

input.value = '';

return;

}

estado[clave].push(valor);

renderTags(clave);

input.value = '';

input.focus();

await guardar(clave);

}


// ── Eliminar ──────────────────────────────────────────────

async function eliminarItem(clave, index) {

if (isViewer()) return;

if (!confirm(`¿Eliminar "${estado[clave][index]}"?`)) return;

estado[clave].splice(index, 1);

renderTags(clave);

await guardar(clave);

}


// ── Guardar ───────────────────────────────────────────────

async function guardar(clave) {

const { error } = await _sb

.from('configuracion')

.upsert({ clave, valor: JSON.stringify(estado[clave]) }, { onConflict: 'clave' });

mostrarToast(error ? '❌ Error al guardar' : '✅ Guardado');

if (error) console.error(error);

}


// ── Tabs ──────────────────────────────────────────────────

function switchTab(nombre) {

document.getElementById('panel-marcas').style.display = nombre === 'marcas' ? 'block' : 'none';

document.getElementById('panel-tratamientos').style.display = nombre === 'tratamientos' ? 'block' : 'none';

document.getElementById('tab-btn-marcas').className = 'ajustes-tab' + (nombre === 'marcas' ? ' activo' : '');

document.getElementById('tab-btn-tratamientos').className = 'ajustes-tab' + (nombre === 'tratamientos' ? ' activo' : '');

}


// ── Helpers ───────────────────────────────────────────────

function escH(s) { return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;'); }


let toastTimer;

function mostrarToast(msg) {

const t = document.getElementById('toast');

t.textContent = msg;

t.classList.add('show');

clearTimeout(toastTimer);

toastTimer = setTimeout(() => t.classList.remove('show'), 2200);

}


cargarTodo();

</script>
