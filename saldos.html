<script>

if (!sessionStorage.getItem('olvision_auth')) location.href = 'index.html';

const { createClient } = supabase;

const db = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);


let todosLosSaldos = [], saldosFiltrados = [], sortMode = 'fecha', registroActual = null;

let empleados = [];


// Fecha hoy para el input

const _hoy = new Date();

const _fechaHoy = `${_hoy.getFullYear()}-${String(_hoy.getMonth()+1).padStart(2,'0')}-${String(_hoy.getDate()).padStart(2,'0')}`;


function salir() { sessionStorage.clear(); location.href = 'index.html'; }

function formatPesos(n) { if(!n||isNaN(n)) return '$0'; return '$'+Math.round(n).toLocaleString('es-AR'); }

function formatFecha(f) { if(!f) return '—'; const [y,m,d] = f.split('-'); return `${d}/${m}/${y}`; }


async function cargarEmpleados() {

const { data } = await db.from('empleados').select('nombre').eq('activo', true).order('nombre');

empleados = (data || []).map(e => e.nombre);

const sel = document.getElementById('filtro-vendedor');

empleados.forEach(n => {

const o = document.createElement('option'); o.value = n; o.textContent = n; sel.appendChild(o);

});

}


async function cargarSaldos() {

const cacheKey = 'saldos_pendientes';

const cached = OlvisionCache.get(cacheKey);

let data;

if (cached) {

data = cached;

} else {

const { data: rows, error } = await db

.from('registros').select('*').gt('saldo', 0)

.or('egreso.is.null,egreso.eq.0').order('fecha', { ascending: true });

if (error) {

document.getElementById('lista-saldos').innerHTML = `<div class="empty"><div class="empty-icon">⚠️</div>Error: ${error.message}</div>`;

return;

}

data = rows || [];

OlvisionCache.set(cacheKey, data);

}

todosLosSaldos = data;

actualizarStats();

filtrar();

}


function actualizarStats() {

const total = todosLosSaldos.reduce((s,r) => s+(r.saldo||0), 0);

const clientes = new Set(todosLosSaldos.map(r => r.nombre_cliente)).size;

const hoy = new Date();

const antiguos = todosLosSaldos.filter(r => {

if (!r.fecha) return false;

return (hoy - new Date(r.fecha+'T12:00:00')) / 86400000 > 30;

}).length;

document.getElementById('stat-total').textContent = formatPesos(total);

document.getElementById('stat-cantidad').textContent = clientes;

document.getElementById('stat-antiguos').textContent = antiguos;

document.getElementById('nav-sub').textContent = `${todosLosSaldos.length} orden${todosLosSaldos.length!==1?'es':''} pendiente${todosLosSaldos.length!==1?'s':''}`;

}


function filtrar() {

const busq = document.getElementById('buscador').value.toLowerCase();

const vendedor = document.getElementById('filtro-vendedor').value;

saldosFiltrados = todosLosSaldos.filter(r => {

const matchB = !busq || (r.nombre_cliente||'').toLowerCase().includes(busq);

const matchV = !vendedor || r.vendedor === vendedor;

return matchB && matchV;

});

ordenar(); renderizar();

}


function setSortMode(mode) {

sortMode = mode;

document.getElementById('sort-fecha').classList.toggle('activo', mode==='fecha');

document.getElementById('sort-monto').classList.toggle('activo', mode==='monto');

ordenar(); renderizar();

}


function ordenar() {

if (sortMode === 'fecha') saldosFiltrados.sort((a,b) => new Date(a.fecha+'T12:00:00') - new Date(b.fecha+'T12:00:00'));

else saldosFiltrados.sort((a,b) => (b.saldo||0) - (a.saldo||0));

}


function renderizar() {

const lista = document.getElementById('lista-saldos');

if (saldosFiltrados.length === 0) {

lista.innerHTML = `<div class="empty"><div class="empty-icon">${todosLosSaldos.length===0?'🎉':'🔍'}</div><p>${todosLosSaldos.length===0?'¡Todo cobrado! No hay saldos pendientes':'Sin resultados para esa búsqueda'}</p></div>`;

return;

}

const hoy = new Date();

lista.innerHTML = saldosFiltrados.map(r => {

const dias = r.fecha ? Math.floor((hoy - new Date(r.fecha+'T12:00:00')) / 86400000) : null;

let cardClass = '', badgeClass = '', badgeLabel = '';

if (dias !== null) {

badgeLabel = `⏱ ${dias}d`;

if (dias > 30) { cardClass='urgente'; badgeClass='urgente'; }

else if (dias <= 7) { cardClass='reciente'; badgeClass='reciente'; }

}

return `<div class="saldo-card ${cardClass}">

<div class="sc-header">

<div class="sc-cliente">${r.nombre_cliente||'Sin nombre'}</div>

<div class="sc-badge ${badgeClass}">${badgeLabel}</div>

</div>

<div class="sc-producto">

<span class="tag">${r.producto||'—'}</span>

Orden #${r.numero_orden||'—'}${r.vendedor?` · 👤 ${r.vendedor}`:''}

</div>

<div class="sc-montos">

<div class="monto-item"><div class="m-val">${formatPesos(r.total||0)}</div><div class="m-lbl">Total</div></div>

<div class="monto-item"><div class="m-val">${formatPesos(r.a_cuenta||0)}</div><div class="m-lbl">A cuenta</div></div>

<div class="monto-item pendiente"><div class="m-val">${formatPesos(r.saldo||0)}</div><div class="m-lbl">Resta cobrar</div></div>

</div>

<div class="sc-footer">

<div class="sc-meta">

📅 ${formatFecha(r.fecha)}

${r.forma_pago?` · ${r.forma_pago}`:''}

${r.observacion?`<br>📝 ${r.observacion}`:''}

</div>

<button class="btn-cobrar" onclick="abrirModal(${r.id})">Cobrar</button>

</div>

</div>`;

}).join('');

}


function abrirModal(id) {

registroActual = saldosFiltrados.find(r => r.id === id);

if (!registroActual) return;

document.getElementById('modal-cliente').textContent = registroActual.nombre_cliente || 'Sin nombre';

document.getElementById('modal-info').innerHTML = `

<strong>Producto:</strong> ${registroActual.producto||'—'}<br>

<strong>Orden:</strong> #${registroActual.numero_orden||'—'}<br>

<strong>Total original:</strong> ${formatPesos(registroActual.total||0)}<br>

<strong>Ya pagó:</strong> ${formatPesos(registroActual.a_cuenta||0)}`;

document.getElementById('modal-monto').textContent = formatPesos(registroActual.saldo||0);

document.getElementById('modal-obs').value = '';

document.getElementById('modal-forma').value = 'Efectivo';

document.getElementById('modal-fecha').value = _fechaHoy;

document.getElementById('btn-confirmar').disabled = false;

document.getElementById('btn-confirmar').textContent = '✅ Confirmar cobro';

document.getElementById('overlay-cobrar').classList.add('visible');

}


function cerrarModal() { document.getElementById('overlay-cobrar').classList.remove('visible'); registroActual = null; }

function cerrarModalFuera(e) { if(e.target===document.getElementById('overlay-cobrar')) cerrarModal(); }


async function confirmarCobro() {

if (!registroActual) return;

const btn = document.getElementById('btn-confirmar');

btn.disabled = true; btn.textContent = 'Guardando...';


const formaPago = document.getElementById('modal-forma').value;

const obs = document.getElementById('modal-obs').value.trim();

const fechaInput = document.getElementById('modal-fecha').value;

const [yyyy, mm] = fechaInput.split('-');

const mes = `${yyyy}-${mm}`;


const nuevoReg = {

fecha: fechaInput, nombre_cliente: registroActual.nombre_cliente,

numero_orden: registroActual.numero_orden, producto: 'Saldo',

total: registroActual.saldo, a_cuenta: 0, saldo: 0,

forma_pago: formaPago, observacion: obs || `Cobro saldo orden #${registroActual.numero_orden}`,

egreso: 0, vendedor: registroActual.vendedor, mes

};


const { error: errInsert } = await db.from('registros').insert(nuevoReg);

if (errInsert) {

mostrarToast('Error: ' + errInsert.message, true);

btn.disabled = false; btn.textContent = '✅ Confirmar cobro'; return;

}


await db.from('registros').update({ saldo: 0 }).eq('id', registroActual.id);


OlvisionCache.clear('saldos_pendientes');

OlvisionCache.clear('caja_' + fechaInput);

OlvisionCache.clear('hist_' + mes);


mostrarToast('✅ Cobro registrado');

cerrarModal();

await cargarSaldos();

}


function mostrarToast(msg, error = false) {

const t = document.getElementById('toast');

t.textContent = msg;

t.className = 'toast show' + (error ? ' error' : '');

setTimeout(() => t.classList.remove('show'), 3000);

}


async function init() {

await cargarEmpleados();

await cargarSaldos();

}

init();

</script>
