<script>

if (!sessionStorage.getItem('olvision_auth')) window.location.href = 'index.html';

const { createClient } = supabase;

const db = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);


let empleados = [];


function formatPesos(n) { if(!n||isNaN(n))return'$0'; return'$'+Math.round(n).toLocaleString('es-AR'); }

function salir() { sessionStorage.removeItem('olvision_auth'); window.location.href='index.html'; }


async function cargarEmpleados() {

const { data } = await db.from('empleados').select('*').order('nombre');

empleados = data || [];

renderLista();

}


function renderLista() {

const lista = document.getElementById('lista-empleados');

if (empleados.length === 0) {

lista.innerHTML = '<div class="empty">No hay empleados todavía.<br>Usá el botón de arriba para agregar.</div>';

return;

}

lista.innerHTML = empleados.map(e => `

<div class="emp-card ${e.activo ? '' : 'inactivo'}">

<div style="flex:1">

<div class="emp-nombre">${e.nombre}</div>

<div class="emp-detalle">

${e.tiene_sueldo ? `💰 ${formatPesos(e.sueldo_quincena)}/quincena` : 'Sin sueldo quincenal'}

${e.comision_pct > 0 ? `<br>📊 Comisión: ${e.comision_pct}%` : ''}

</div>

<div class="emp-badges">

<span class="emp-badge ${e.activo ? 'verde' : 'gris'}">${e.activo ? '✅ Activo' : '⛔ Inactivo'}</span>

${e.tiene_sueldo ? '<span class="emp-badge azul">💰 Sueldo</span>' : ''}

${e.comision_pct > 0 ? `<span class="emp-badge celeste">📊 ${e.comision_pct}%</span>` : ''}

</div>

</div>

<div class="emp-acciones">

<button class="emp-btn" style="background:#f1f5f9;color:#475569" onclick="abrirEditar(${e.id})">✏️</button>

<button class="emp-btn" style="background:${e.activo?'#fef2f2':'#f0fdf4'};color:${e.activo?'#dc2626':'#059669'}"

onclick="toggleActivo(${e.id},${e.activo})">${e.activo?'⛔':'✅'}</button>

</div>

</div>`).join('');

}


function abrirNuevo() {

document.getElementById('emp-titulo').textContent = 'Nuevo empleado';

document.getElementById('emp-id').value = '';

document.getElementById('emp-nombre').value = '';

document.getElementById('emp-tiene-sueldo').checked = false;

document.getElementById('emp-sueldo-campo').style.display = 'none';

document.getElementById('emp-sueldo').value = '';

document.getElementById('emp-comision').value = '0';

document.getElementById('emp-btn-guardar').textContent = '✅ Crear empleado';

document.getElementById('overlay-empleado').classList.add('visible');

setTimeout(() => document.getElementById('emp-nombre').focus(), 200);

}


function abrirEditar(id) {

const e = empleados.find(x => x.id === id);

if (!e) return;

document.getElementById('emp-titulo').textContent = `✏️ Editar — ${e.nombre}`;

document.getElementById('emp-id').value = e.id;

document.getElementById('emp-nombre').value = e.nombre;

document.getElementById('emp-tiene-sueldo').checked = e.tiene_sueldo;

document.getElementById('emp-sueldo-campo').style.display = e.tiene_sueldo ? 'block' : 'none';

setVal(document.getElementById('emp-sueldo'), e.sueldo_quincena);

document.getElementById('emp-comision').value = e.comision_pct;

document.getElementById('emp-btn-guardar').textContent = '💾 Guardar cambios';

document.getElementById('overlay-empleado').classList.add('visible');

}


function cerrarModal() { document.getElementById('overlay-empleado').classList.remove('visible'); }

function cerrarFuera(e) { if (e.target === document.getElementById('overlay-empleado')) cerrarModal(); }

function toggleSueldoField() {

document.getElementById('emp-sueldo-campo').style.display =

document.getElementById('emp-tiene-sueldo').checked ? 'block' : 'none';

}

function toggleCheck() {

const cb = document.getElementById('emp-tiene-sueldo');

cb.checked = !cb.checked; toggleSueldoField();

}


async function guardarEmpleado() {

const id = document.getElementById('emp-id').value;

const nombre = document.getElementById('emp-nombre').value.trim();

const tiene_sueldo = document.getElementById('emp-tiene-sueldo').checked;

const sueldo_quincena = tiene_sueldo ? (getVal(document.getElementById('emp-sueldo')) || 0) : 0;

const comision_pct = parseFloat(document.getElementById('emp-comision').value) || 0;

if (!nombre) { alert('Ingresá el nombre'); return; }

const btn = document.getElementById('emp-btn-guardar');

btn.textContent = 'Guardando...'; btn.disabled = true;

let error;

if (id) {

({ error } = await db.from('empleados').update({ nombre, tiene_sueldo, sueldo_quincena, comision_pct }).eq('id', id));

} else {

({ error } = await db.from('empleados').insert([{ nombre, tiene_sueldo, sueldo_quincena, comision_pct, activo: true }]));

}

if (error) { alert('Error: ' + error.message); btn.textContent = id ? '💾 Guardar cambios' : '✅ Crear empleado'; btn.disabled = false; return; }

cerrarModal();

await cargarEmpleados();

btn.textContent = id ? '💾 Guardar cambios' : '✅ Crear empleado';

btn.disabled = false;

}


async function toggleActivo(id, activo) {

const e = empleados.find(x => x.id === id);

const accion = activo ? 'desactivar' : 'activar';

if (!confirm(`¿${accion.charAt(0).toUpperCase()+accion.slice(1)} a ${e?.nombre}?\n\n${activo ? 'Desaparece de toda la app.' : 'Vuelve a aparecer en toda la app.'}`)) return;

await db.from('empleados').update({ activo: !activo }).eq('id', id);

await cargarEmpleados();

}


cargarEmpleados();
