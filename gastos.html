function getMesStr() { return `${anioActual}-${String(mesActual+1).padStart(2,'0')}`; }

function getMesSiguienteStr(){ if(mesActual===11) return `${anioActual+1}-01-01`; return `${anioActual}-${String(mesActual+2).padStart(2,'0')}-01`; }

function formatPesos(n) { if(!n||isNaN(n)) return '$0'; return '$'+Math.round(n).toLocaleString('es-AR'); }

function formatFecha(f) { if(!f) return ''; const[y,m,d]=f.split('-'); return`${d}/${m}/${y}`; }

function cambiarMes(dir) {

mesActual += dir;

async function cargar() {

const mes = getMesStr(), cacheKey = 'gastos_' + mes;

const cached = OlvisionCache.get(cacheKey);

if(cached){ gastos = cached; renderTodo(); }

const { data } = await db.from('gastos').select('*')

.gte('fecha', mes+'-01').lt('fecha', getMesSiguienteStr())

.order('fecha', { ascending: false });

gastos = data || [];

OlvisionCache.set(cacheKey, gastos);

renderTodo();

}

async function exportarPDFGastos() {

if (!gastos.length) { alert('No hay gastos este mes para exportar.'); return; }

const btn = document.getElementById('btn-pdf');

btn.textContent = '⏳ PDF...'; btn.disabled = true;

try {

const { jsPDF } = window.jspdf;

prepararTemplate();

const el = document.getElementById('reporte-img');

const canvas = await html2canvas(el, { scale: 2, backgroundColor: '#ffffff', useCORS: true, logging: false });

const imgData = canvas.toDataURL('image/png');

const pw = 210, ph = 297; // A4 portrait, en mm

const imgH = (canvas.height * pw) / canvas.width;

const pdf = new jsPDF({ orientation: 'p', unit: 'mm', format: 'a4' });

// Paginar si el contenido es más largo que una página

let yOffset = 0, remaining = imgH, page = 0;

while (remaining > 0) {

if (page > 0) pdf.addPage();

pdf.addImage(imgData, 'PNG', 0, -yOffset, pw, imgH);

yOffset += ph;

remaining -= ph;

page++;

}

pdf.save(`gastos_${getMesStr()}.pdf`);

} catch(e) { alert('Error al generar PDF: ' + e.message); }

btn.textContent = '📄 PDF'; btn.disabled = false;

}
