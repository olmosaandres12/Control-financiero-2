let mensaje = `💼 Olvisión — Sueldo ${empleada}\n\n`;

let totalGeneral = 0, pagadoGeneral = 0;


[1, 2].forEach(q => {

const sd = sueldosEmpleada.find(s => s.quincena === q);

if (!sd) return;

const total = parseFloat(sd.monto_total) || 0;

totalGeneral += total;

const qPagos = todosPagos.filter(p => String(p.sueldo_id) === String(sd.id));

pagadoGeneral += qPagos.reduce((s, p) => s + (parseFloat(p.monto) || 0), 0);

const qTag = q === 1 ? '(1Q)' : '(2Q)';

const nroLabel = q === 1 ? '1ra' : '2da';

mensaje += `${div}\n`;

mensaje += `${nroLabel} quincena · ${formatPesos(total)}\n`;

if (qPagos.length === 0) {

mensaje += ` Sin pagos registrados\n`;

} else {

qPagos.forEach(p => {

const esHoy = p.id === waPayId;

mensaje += ` 📅 ${formatFecha(p.fecha)} ${qTag} ${formatPesos(p.monto)} ${p.forma_pago || ''}${esHoy ? ' ← hoy' : ''}\n`;

});

}

});


const saldoGeneral = totalGeneral - pagadoGeneral;

mensaje += `${div}\n`;

if (saldoGeneral <= 0) {

mensaje += `Pagado: ${formatPesos(pagadoGeneral)}\n`;

mensaje += '✅ Mes saldado';

} else {

mensaje += `Pagado: ${formatPesos(pagadoGeneral)}\n`;

mensaje += `Saldo: ${formatPesos(saldoGeneral)}`;

}


document.getElementById('wa-texto').textContent = mensaje;

document.getElementById('wa-copiado').style.display = 'none';

document.getElementById('overlay-wa').classList.add('visible');

}


function copiarMensajeWA() {

const texto = document.getElementById('wa-texto').textContent;

const mostrarOk = () => {

const aviso = document.getElementById('wa-copiado');

aviso.style.display = 'block';

setTimeout(() => { aviso.style.display = 'none'; }, 2500);

};

if (navigator.clipboard) {

navigator.clipboard.writeText(texto).then(mostrarOk).catch(() => fallbackCopy(texto, mostrarOk));

} else {

fallbackCopy(texto, mostrarOk);

}

}


function fallbackCopy(texto, cb) {

const ta = document.createElement('textarea');

ta.value = texto; ta.style.cssText = 'position:fixed;opacity:0;top:0;left:0';

document.body.appendChild(ta); ta.focus(); ta.select();

try { document.execCommand('copy'); cb(); } catch(e) {}

document.body.removeChild(ta);

}

async function eliminarPago(pagoId,sueldoId) {

if(!confirm('¿Eliminar este pago?\n\nNota: el gasto en Gastos NO se elimina automáticamente.'))return;

const pago=pagos.find(p=>p.id===pagoId),sueldo=sueldos.find(s=>s.id===sueldoId);

if(!pago||!sueldo)return;

await db.from('pagos_sueldos').delete().eq('id',pagoId);

const nMP=Math.max(0,(parseFloat(sueldo.monto_pagado)||0)-(parseFloat(pago.monto)||0));

const nS=(parseFloat(sueldo.monto_total)||0)-nMP;

await db.from('sueldos').update({monto_pagado:nMP,saldo:nS}).eq('id',sueldoId);

OlvisionCache.clear('sueldos_'+getPeriodo());await cargar();

}


async function detectarMesConDeuda() {

const hoy=new Date();let meses=[];

for(let i=3;i>=0;i--){const f=new Date(hoy.getFullYear(),hoy.getMonth()-i,1);meses.push(`${f.getFullYear()}-${String(f.getMonth()+1).padStart(2,'0')}`);}

const{data}=await db.from('sueldos').select('periodo,saldo').in('periodo',meses).gt('saldo',0).order('periodo',{ascending:true});

if(data&&data.length>0){const[y,m]=data[0].periodo.split('-');mesActual=parseInt(m)-1;anioActual=parseInt(y);}

}


async function renderAlertaMeses() {

const hoy=new Date();let meses=[];

for(let i=3;i>=0;i--){const f=new Date(hoy.getFullYear(),hoy.getMonth()-i,1);meses.push(`${f.getFullYear()}-${String(f.getMonth()+1).padStart(2,'0')}`);}

const{data}=await db.from('sueldos').select('periodo,saldo,empleada').in('periodo',meses).gt('saldo',0);

if(!data||data.length===0)return;

const porP={};data.forEach(s=>{if(!porP[s.periodo])porP[s.periodo]=[];porP[s.periodo].push(s);});

const periodoActual=getPeriodo();

const mesesDeuda=Object.keys(porP).sort().filter(p=>p!==periodoActual);

if(mesesDeuda.length===0)return;

const banner=document.createElement('div');

banner.id='banner-deuda';

banner.style.cssText='background:#fef2f2;border:2px solid #ef4444;border-radius:16px;padding:14px 16px;margin-bottom:16px;';

banner.innerHTML=`<div style="font-size:13px;font-weight:800;color:#991b1b;margin-bottom:8px">⚠️ Meses con sueldos pendientes</div><div style="display:flex;flex-wrap:wrap;gap:8px">${mesesDeuda.map(p=>{const[y,m]=p.split('-');const tot=porP[p].reduce((s,x)=>s+(parseFloat(x.saldo)||0),0);return`<button onclick="irAlMes('${p}')" style="background:#ef4444;color:white;border:none;border-radius:10px;padding:8px 14px;font-size:13px;font-weight:800;cursor:pointer;font-family:'Nunito',sans-serif">${MESES[parseInt(m)-1]} ${y} — $${Math.round(tot).toLocaleString('es-AR')}</button>`;}).join('')}</div>`;

const panel=document.getElementById('panel-sueldos');

panel.insertBefore(banner,panel.querySelector('.periodo-selector'));

}
