<script>

if (!sessionStorage.getItem('olvision_auth')) window.location.href = 'index.html';

const { createClient } = supabase;

const db = createClient(SUPABASE_URL, SUPABASE_ANON_KEY);


function salir() { sessionStorage.removeItem('olvision_auth'); window.location.href='index.html'; }


// ── Mapeo de meses ───────────────────────────────────────

const MESES_MAP = {

'ENE':'01','FEB':'02','MAR':'03','ABR':'04','MAY':'05','JUN':'06',

'JUL':'07','AGO':'08','SET':'09','SEP':'09','OCT':'10','NOV':'11','DIC':'12'

};

const MESES_LABEL = {

'01':'Enero','02':'Febrero','03':'Marzo','04':'Abril','05':'Mayo','06':'Junio',

'07':'Julio','08':'Agosto','09':'Septiembre','10':'Octubre','11':'Noviembre','12':'Diciembre'

};


let workbook = null;

let datosParaImportar = []; // array de registros listos para Supabase


// ── Drag and drop ────────────────────────────────────────

const dropZone = document.getElementById('drop-zone');

dropZone.addEventListener('dragover', e => { e.preventDefault(); dropZone.classList.add('dragover'); });

dropZone.addEventListener('dragleave', () => dropZone.classList.remove('dragover'));

dropZone.addEventListener('drop', e => {

e.preventDefault(); dropZone.classList.remove('dragover');

const file = e.dataTransfer.files[0];

if (file) procesarArchivo(file);

});


function leerArchivo(input) {

if (input.files[0]) procesarArchivo(input.files[0]);

}


function procesarArchivo(file) {

if (!file.name.match(/\.(xlsx|xls)$/i)) {

alert('Solo se aceptan archivos .xlsx o .xls');

return;

}

const reader = new FileReader();

reader.onload = e => {

try {

workbook = XLSX.read(e.target.result, { type: 'array', cellDates: true });

dropZone.classList.add('tiene-archivo');

document.getElementById('dz-icon').textContent = '✅';

document.getElementById('dz-texto').textContent = file.name;

document.getElementById('dz-sub').textContent = 'Archivo listo · ' + (file.size/1024).toFixed(0) + ' KB';

document.getElementById('btn-preview').disabled = false;

} catch(err) {

alert('Error al leer el archivo: ' + err.message);

}

};

reader.readAsArrayBuffer(file);

}


// ── Generar preview ──────────────────────────────────────

function generarPreview() {

if (!workbook) return;

const anio = document.getElementById('sel-anio').value;

datosParaImportar = [];


const sheetNames = workbook.SheetNames;

const mesesEncontrados = sheetNames.filter(n => MESES_MAP[n.toUpperCase()]);


if (mesesEncontrados.length === 0) {

alert('No encontré hojas de meses (ENE, FEB, MAR...) en el archivo.');

return;

}


let totalOps = 0;

let totalMonto = 0;

const resumenMeses = [];


mesesEncontrados.forEach(nombreHoja => {

const mesNum = MESES_MAP[nombreHoja.toUpperCase()];

const mesClave = `${anio}-${mesNum}`;

const ws = workbook.Sheets[nombreHoja];

const filas = XLSX.utils.sheet_to_json(ws, { header: 1, defval: '' });


// Buscar fila de encabezado (primera con "FECHA" o "NOMBRE")

let filaHeader = 0;

for (let i = 0; i < Math.min(5, filas.length); i++) {

const row = filas[i].map(c => String(c).toUpperCase().trim());

if (row.includes('FECHA') || row.includes('NOMBRE')) {

filaHeader = i;

break;

}

}


const registrosMes = [];


for (let i = filaHeader + 1; i < filas.length; i++) {

const row = filas[i];

if (!row || row.length < 5) continue;


const fechaVal = row[0];

const nombreVal = String(row[1] || '').trim();

const ordenVal = String(row[2] || '').trim();

const productoVal = String(row[3] || '').trim();

const totalVal = parseFloat(row[4]) || 0;

const aCuentaVal = parseFloat(row[5]) || 0;

const saldoVal = parseFloat(row[6]) || 0;

const formaPagoVal = String(row[7] || '').trim();

const observacionVal = String(row[8] || '').trim();

const egresoVal = parseFloat(row[9]) || 0;

const vendedorVal = String(row[10] || '').trim();


// Ignorar filas de cierre (🏁) o vacías

const fechaStr = String(fechaVal);

if (fechaStr.includes('🏁') || fechaStr.includes('CIERRE')) continue;

if (!fechaVal && totalVal === 0 && egresoVal === 0) continue;

if (totalVal === 0 && egresoVal === 0) continue;


// Parsear fecha

let fechaISO = null;

if (fechaVal instanceof Date) {

fechaISO = fechaVal.toISOString().split('T')[0];

} else if (typeof fechaVal === 'number') {

// Excel serial date

const d = XLSX.SSF.parse_date_code(fechaVal);

if (d) {

fechaISO = `${anio}-${String(d.m).padStart(2,'0')}-${String(d.d).padStart(2,'0')}`;

}

} else if (typeof fechaVal === 'string' && fechaVal.trim()) {

// Formato dd/MM o dd/MM/yyyy

const partes = fechaVal.trim().split('/');

if (partes.length >= 2) {

const dd = partes[0].padStart(2,'0');

const mm = partes[1].padStart(2,'0');

const yyyy = partes.length >= 3 ? partes[2] : anio;

fechaISO = `${yyyy}-${mm}-${dd}`;

}

}


if (!fechaISO) continue;

// Validar que el mes del ISO coincide con la hoja

const mesFecha = fechaISO.substring(5,7);

if (mesFecha !== mesNum) continue;


registrosMes.push({

fecha: fechaISO,

nombre_cliente: nombreVal || null,

numero_orden: ordenVal || null,

producto: productoVal || null,

total: totalVal,

a_cuenta: aCuentaVal,

saldo: saldoVal,

forma_pago: formaPagoVal || null,

observacion: observacionVal || null,

egreso: egresoVal,

vendedor: vendedorVal || null,

mes: mesClave,

});

}


const montoMes = registrosMes.reduce((s,r) => s + (r.egreso > 0 ? 0 : (r.a_cuenta > 0 ? r.a_cuenta : r.total)), 0);

resumenMeses.push({ nombre: nombreHoja, mes: mesClave, cant: registrosMes.length, monto: montoMes });

totalOps += registrosMes.length;

totalMonto += montoMes;

datosParaImportar.push(...registrosMes);

});


if (totalOps === 0) {

alert('No se encontraron operaciones válidas en el archivo.');

return;

}


// Renderizar preview

document.getElementById('preview-sub').textContent =

`${totalOps} operaciones en ${mesesEncontrados.length} meses — Total: $${Math.round(totalMonto).toLocaleString('es-AR')}`;


const lista = document.getElementById('preview-lista');

lista.innerHTML = resumenMeses.map(m => `

<div class="preview-mes ${m.cant > 0 ? 'tiene-datos' : 'sin-datos'}">

<div>

<div class="preview-mes-nombre">${MESES_LABEL[m.mes.split('-')[1]] || m.nombre} ${anio}</div>

<div class="preview-mes-count">${m.cant} operaciones</div>

</div>

<div class="preview-mes-monto">${m.cant > 0 ? '$' + Math.round(m.monto).toLocaleString('es-AR') : '—'}</div>

</div>

`).join('');


// Verificar si ya hay datos de estos meses en Supabase

verificarDuplicados(resumenMeses.filter(m => m.cant > 0).map(m => m.mes));


mostrarStep('step-2');

}


async function verificarDuplicados(meses) {

const alertaDiv = document.getElementById('alerta-dup');

try {

const { data } = await db.from('registros')

.select('mes')

.in('mes', meses)

.limit(1);


if (data && data.length > 0) {

alertaDiv.style.display = 'block';

alertaDiv.textContent = '⚠️ Ya existen registros de algunos de estos meses en la base de datos. Si importás, pueden aparecer duplicados. Verificá antes de continuar.';

} else {

alertaDiv.style.display = 'none';

}

} catch(e) { alertaDiv.style.display = 'none'; }

}


// ── Importar ─────────────────────────────────────────────

async function importar() {

if (datosParaImportar.length === 0) return;

mostrarStep('step-3');


const log = document.getElementById('log-box');

const progFill = document.getElementById('prog-fill');

const progTexto = document.getElementById('prog-texto');

const progPct = document.getElementById('prog-pct');


let importados = 0;

let errores = [];

const BATCH = 50; // insertar de a 50 registros

const total = datosParaImportar.length;


agregarLog(log, 'info', `Iniciando importación de ${total} registros...`);


for (let i = 0; i < datosParaImportar.length; i += BATCH) {

const lote = datosParaImportar.slice(i, i + BATCH);

const mesActual = lote[0].mes;


try {

const { error } = await db.from('registros').insert(lote);

if (error) {

errores.push(`Lote ${Math.floor(i/BATCH)+1}: ${error.message}`);

agregarLog(log, 'err', `❌ Error en lote (${mesActual}): ${error.message}`);

} else {

importados += lote.length;

agregarLog(log, 'ok', `✅ ${lote.length} registros de ${mesActual} importados`);

}

} catch(e) {

errores.push(e.message);

agregarLog(log, 'err', `❌ ${e.message}`);

}


const pct = Math.round(((i + lote.length) / total) * 100);

progFill.style.width = pct + '%';

progPct.textContent = pct + '%';

progTexto.textContent = `Importando ${mesActual}...`;


// Pequeña pausa para no saturar Supabase

await new Promise(r => setTimeout(r, 100));

}


agregarLog(log, 'info', `Finalizado: ${importados} importados, ${errores.length} errores`);


// Mostrar resultado

document.getElementById('res-num').textContent = importados;

if (errores.length > 0) {

document.getElementById('res-errores').innerHTML =

`<div class="resultado-error">⚠️ ${errores.length} error(es):<br>${errores.join('<br>')}</div>`;

}


mostrarStep('step-4');

}


function agregarLog(logBox, tipo, texto) {

const div = document.createElement('div');

div.className = tipo === 'ok' ? 'log-ok' : tipo === 'err' ? 'log-err' : 'log-info';

div.textContent = texto;

logBox.appendChild(div);

logBox.scrollTop = logBox.scrollHeight;

}


function mostrarStep(id) {

document.querySelectorAll('.step').forEach(s => s.classList.remove('visible'));

document.getElementById(id).classList.add('visible');

}


function resetear() {

workbook = null;

datosParaImportar = [];

document.getElementById('file-input').value = '';

document.getElementById('dz-icon').textContent = '📊';

document.getElementById('dz-texto').textContent = 'Tocá para elegir el archivo';

document.getElementById('dz-sub').textContent = 'o arrastralo acá · .xlsx, .xls';

document.getElementById('drop-zone').classList.remove('tiene-archivo');

document.getElementById('btn-preview').disabled = true;

document.getElementById('prog-fill').style.width = '0%';

document.getElementById('log-box').innerHTML = '';

document.getElementById('res-errores').innerHTML = '';

mostrarStep('step-1');

}
